# Re-enable legacy KEX, host keys, and ciphers for older Cisco IOS devices (requires paramiko==3.5.1)
for kex in (
    "diffie-hellman-group14-sha1",
    "diffie-hellman-group-exchange-sha1",
    "diffie-hellman-group1-sha1",
):
    if kex not in paramiko.Transport._preferred_kex:
        paramiko.Transport._preferred_kex += (kex,)

for key in ("ssh-rsa", "rsa-sha2-256", "rsa-sha2-512"):
    if key not in paramiko.Transport._preferred_keys:
        paramiko.Transport._preferred_keys += (key,)

for cipher in ("aes128-cbc", "aes256-cbc", "3des-cbc"):
    if cipher not in paramiko.Transport._preferred_ciphers:
        paramiko.Transport._preferred_ciphers += (cipher,)

def verify_admin(credentials: HTTPBasicCredentials = Depends(security)):
    admin_user = os.getenv("ADMIN_USER", "admin")
    admin_pass = os.getenv("ADMIN_PASSWORD", "")
    if not admin_pass:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="ADMIN_PASSWORD is not configured in .env",
        )
    correct_user = secrets.compare_digest(credentials.username, admin_user)
    correct_pass = secrets.compare_digest(credentials.password, admin_pass)
    if not (correct_user and correct_pass):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid admin credentials",
            headers={"WWW-Authenticate": "Basic"},
        )
    return credentials.username

class CheckRequest(BaseModel):

    username: str
    password: str
    secret: str = ""
    device_type: str = "switch"  # 'switch' or 'router'
    phase: str = "PRE"           # 'PRE' (Before) or 'POST' (After)


class UpgradeRequest(BaseModel):
    host: str
    username: str
    password: str
    secret: str = ""
    device_type: str = "switch"
    image_filename: str
    expected_md5: str = ""

    auto_clean_flash: bool = False
    auto_pre_check: bool = True
    auto_post_check: bool = True


def resolve_vendor(mac_addr: str) -> str:
    try:
        clean = mac_addr.replace(".", "").replace(":", "").replace("-", "").strip()
        if len(clean) < 12 or clean.lower() == "incomplete":
            return "Incomplete / Unknown"
        mac = EUI(clean)
        return mac.oui.registration().org
    except NotRegisteredError:
        return "Unknown / Locally Administered"
    except Exception:
        return "Unknown / Locally Administered"


def compute_md5(filepath: str) -> str:
    h = hashlib.md5()
    with open(filepath, "rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def build_conn_params(host: str, username: str, password: str, secret: str = "") -> dict:
    return {
        "device_type": "cisco_ios",
        "host": host,
        "username": username,
        "password": password,
        "secret": secret or password,
        "disabled_algorithms": {
            "pubkeys": [],
            "kex": [],
            "ciphers": [],
            "macs": [],
        },
    }


def parse_flash_free_bytes(dir_output: str) -> int:
    """Extract free bytes from 'dir flash:' output across Cisco IOS/IOS-XE formats."""
    match = re.search(r"\((\d+)\s+bytes\s+free\)", dir_output, re.IGNORECASE)
    if match:
        return int(match.group(1))
    match_alt = re.search(r"(\d+)\s+bytes\s+available", dir_output, re.IGNORECASE)
    if match_alt:
        return int(match_alt.group(1))
    return -1


def check_remote_file_exists(net_connect, remote_filename: str) -> bool:
    out = net_connect.send_command(f"dir flash:{remote_filename}")
    if "no such file" in out.lower() or "error opening" in out.lower() or "%error" in out.lower():
        return False
    return remote_filename.lower() in out.lower()


def run_install_remove_inactive(net_connect, job_id: int):
    append_job_log(job_id, "Running 'install remove inactive' to reclaim flash space...")
    out = net_connect.send_command_timing(
        "install remove inactive", strip_prompt=False, strip_command=False, read_timeout=300
    )
    for _ in range(4):
        if "[y/n]" in out.lower() or "[yes/no]" in out.lower() or "proceed" in out.lower():
            ans = "yes" if "[yes/no]" in out.lower() else "y"
            out = net_connect.send_command_timing(
                ans, strip_prompt=False, strip_command=False, read_timeout=300
            )
        else:
            break
    append_job_log(job_id, "Inactive package cleanup completed.")


def collect_and_save_snapshot(host: str, username: str, password: str, secret: str, device_type: str, phase: str) -> dict:
    conn_params = build_conn_params(host, username, password, secret)
    with ConnectHandler(**conn_params) as net_connect:
        net_connect.enable()
        prompt_hostname = net_connect.find_prompt().rstrip("#>") or host

        interfaces = net_connect.send_command("show interface status", use_textfsm=True)
        if not isinstance(interfaces, list) or len(interfaces) == 0:
            interfaces = net_connect.send_command("show ip interface brief", use_textfsm=True)
        if not isinstance(interfaces, list):
            interfaces = []

        raw_arp = net_connect.send_command("show ip arp", use_textfsm=True)
        arp_table = []
        if isinstance(raw_arp, list):
            for entry in raw_arp:
                mac = entry.get("mac") or entry.get("mac_address") or ""
                entry["vendor"] = resolve_vendor(mac)
                arp_table.append(entry)

        mac_table = []
        if device_type.lower() == "switch":
            raw_macs = net_connect.send_command("show mac address-table", use_textfsm=True)
            if isinstance(raw_macs, list):
                for entry in raw_macs:
                    raw_port = entry.get("destination_port") or entry.get("ports") or ""
                    port_str = ",".join(raw_port) if isinstance(raw_port, list) else str(raw_port)
                    if "cpu" in port_str.lower():
                        continue
                    if "vlan" not in entry and "vlan_id" in entry:
                        entry["vlan"] = entry["vlan_id"]
                    mac = entry.get("destination_address") or entry.get("mac", "")
                    entry["vendor"] = resolve_vendor(mac)
                    mac_table.append(entry)

        running_config = net_connect.send_command("show running-config")

    conn = get_db()
    cur = conn.cursor()
    cur.execute(
        """
        INSERT INTO devices (hostname, ip_address, device_type)
        VALUES (%s, %s, %s)
        ON CONFLICT (ip_address) DO UPDATE
        SET hostname = EXCLUDED.hostname, device_type = EXCLUDED.device_type
        RETURNING id;
        """,
        (prompt_hostname, host, device_type.lower()),
    )
    device_id = cur.fetchone()["id"]

    cur.execute(
        """
        INSERT INTO health_snapshots
        (device_id, phase, interfaces_json, arp_table_json, mac_table_json, running_config)
        VALUES (%s, %s, %s, %s, %s, %s)
        RETURNING id, captured_at;
        """,
        (
            device_id,
            phase.upper(),
            json.dumps(interfaces),
            json.dumps(arp_table),
            json.dumps(mac_table),
            running_config,
        ),
    )
    snap = cur.fetchone()
    conn.commit()
    cur.close()
    conn.close()

    return {
        "device_id": device_id,
        "hostname": prompt_hostname,
        "phase": phase.upper(),
        "timestamp_pst": snap["captured_at"].astimezone(PST).strftime("%Y-%m-%d %H:%M:%S %Z"),
    }


def append_job_log(job_id: int, message: str, new_status: str = None, hostname: str = None):
    ts = datetime.now(PST).strftime("%Y-%m-%d %H:%M:%S %Z")
    line = f"[{ts}] {message}\n"
    conn = get_db()
    cur = conn.cursor()
    if new_status in ("COMPLETED", "FAILED"):
        cur.execute(
            """
            UPDATE upgrade_jobs
            SET logs = logs || %s, status = %s, hostname = COALESCE(%s, hostname), finished_at = NOW()
            WHERE id = %s;
            """,
            (line, new_status, hostname, job_id),
        )
    elif new_status:
        cur.execute(
            """
            UPDATE upgrade_jobs
            SET logs = logs || %s, status = %s, hostname = COALESCE(%s, hostname)
            WHERE id = %s;
            """,
            (line, new_status, hostname, job_id),
        )
    else:
        cur.execute(
            "UPDATE upgrade_jobs SET logs = logs || %s, hostname = COALESCE(%s, hostname) WHERE id = %s;",
            (line, hostname, job_id),
        )
    conn.commit()
    cur.close()
    conn.close()


def ensure_image_staged_on_flash(net_connect, job_id: int, image_path: str, remote_filename: str, auto_clean_flash: bool) -> bool:
    image_size_bytes = os.path.getsize(image_path)
    image_size_mb = round(image_size_bytes / (1024 * 1024), 2)
    # Require 1.5x image size (or at least image + 250MB) for IOS-XE package expansion
    required_bytes = max(int(image_size_bytes * 1.5), image_size_bytes + (250 * 1024 * 1024))
    required_mb = round(required_bytes / (1024 * 1024), 2)

    already_on_flash = check_remote_file_exists(net_connect, remote_filename)
    if already_on_flash:
        append_job_log(job_id, f"Pre-flight check: {remote_filename} already exists on flash:. Verifying MD5...")
    else:
        append_job_log(job_id, f"Pre-flight flash space check (Image: {image_size_mb} MB | Recommended Free: {required_mb} MB)...")
        dir_out = net_connect.send_command("dir flash:")
        free_bytes = parse_flash_free_bytes(dir_out)

        if free_bytes >= 0:
            free_mb = round(free_bytes / (1024 * 1024), 2)
            append_job_log(job_id, f"Detected {free_mb} MB free on flash:.")

            if free_bytes < required_bytes:
                if auto_clean_flash:
                    append_job_log(job_id, f"Free space ({free_mb} MB) is below recommended {required_mb} MB. Auto-cleanup enabled...")
                    run_install_remove_inactive(net_connect, job_id)
                    dir_out = net_connect.send_command("dir flash:")
                    free_bytes = parse_flash_free_bytes(dir_out)
                    free_mb = round(free_bytes / (1024 * 1024), 2)
                    append_job_log(job_id, f"Post-cleanup free space on flash:: {free_mb} MB.")

                if free_bytes < required_bytes:
                    append_job_log(
                        job_id,
                        f"ERROR: Insufficient flash space ({free_mb} MB free, {required_mb} MB required for transfer + package expansion). Aborting.",
                        "FAILED",
                    )
                    return False
        else:
            append_job_log(job_id, "WARNING: Could not parse free bytes from 'dir flash:'. Proceeding with SCP transfer...")

        append_job_log(job_id, f"Transferring {remote_filename} ({image_size_mb} MB) to flash: via SCP...")

    transfer_res = file_transfer(
        net_connect,
        source_file=image_path,
        dest_file=remote_filename,
        file_system="flash:",
        direction="put",
        overwrite_file=False,
        verify_file=True,
    )
    if not transfer_res.get("file_verified"):
        append_job_log(job_id, "ERROR: Remote MD5 verification failed on flash:!", "FAILED")
        return False

    append_job_log(job_id, f"Verified {remote_filename} on flash: (MD5 matches local image).")
    return True


def execute_iosxe_upgrade(job_id: int, req: UpgradeRequest):
    mode = req.mode.upper()
    if mode not in ("STAGE_ONLY", "ACTIVATE_ONLY", "FULL_UPGRADE"):
        append_job_log(job_id, f"ERROR: Unsupported mode '{mode}'.", "FAILED")
        return

    remote_filename = os.path.basename(req.image_filename)
    image_path = os.path.join(IMAGES_DIR, remote_filename)
    if not os.path.isfile(image_path):
        append_job_log(job_id, f"ERROR: Image file {remote_filename} not found in /app/images.", "FAILED")
        return

    try:
        # 1. Verify Local MD5
        append_job_log(job_id, f"Calculating local MD5 for {remote_filename}...")
        local_md5 = compute_md5(image_path)
        append_job_log(job_id, f"Local MD5: {local_md5}")

        if req.expected_md5 and req.expected_md5.strip().lower() != local_md5.lower():
            append_job_log(
                job_id,
                f"ERROR: MD5 mismatch! Expected {req.expected_md5.strip().lower()}, got {local_md5}",
                "FAILED",
            )
            return

        # 2. Automatic PRE-upgrade Health Check (only when rebooting/upgrading)
        if mode in ("ACTIVATE_ONLY", "FULL_UPGRADE") and req.auto_pre_check:
            append_job_log(job_id, "Running automatic PRE-upgrade health check snapshot...")
            pre_res = collect_and_save_snapshot(
                req.host, req.username, req.password, req.secret, req.device_type, "PRE"
            )
            append_job_log(job_id, f"PRE snapshot saved for {pre_res['hostname']}.", hostname=pre_res["hostname"])

        # 3. Connect to Device & Ensure Image is Staged + Verified on flash:
        conn_params = build_conn_params(req.host, req.username, req.password, req.secret)
        with ConnectHandler(**conn_params) as net_connect:
            net_connect.enable()
            hostname = net_connect.find_prompt().rstrip("#>") or req.host
            append_job_log(job_id, f"Connected to {hostname}.", hostname=hostname)

            ver_out = net_connect.send_command("show version | include Cisco IOS")
            append_job_log(job_id, f"Current Version: {ver_out.strip()}")

            staged_ok = ensure_image_staged_on_flash(
                net_connect, job_id, image_path, remote_filename, req.auto_clean_flash
            )
            if not staged_ok:
                return

            # If STAGE_ONLY, stop here cleanly with zero reboot
            if mode == "STAGE_ONLY":
                append_job_log(
                    job_id,
                    f"STAGING COMPLETE: {remote_filename} is verified on flash: and ready for future activation. No reboot performed.",
                    "COMPLETED",
                )
                return

            # 4. Save Config & Activate/Commit IOS-XE Install Mode
            append_job_log(job_id, "Saving running configuration (write memory)...")
            net_connect.send_command("write memory")

            install_cmd = f"install add file flash:{remote_filename} activate commit"
            append_job_log(job_id, f"Executing: {install_cmd}")

            output = net_connect.send_command_timing(
                install_cmd, strip_prompt=False, strip_command=False, read_timeout=300
            )
            append_job_log(job_id, output.strip())

            for _ in range(4):
                if "[y/n]" in output.lower() or "[yes/no]" in output.lower() or "proceed" in output.lower():
                    answer = "yes" if "[yes/no]" in output.lower() else "y"
                    append_job_log(job_id, f"Sending confirmation '{answer}'...")
                    try:
                        output = net_connect.send_command_timing(
                            answer, strip_prompt=False, strip_command=False, read_timeout=300
                        )
                        append_job_log(job_id, output.strip())
                    except Exception:
                        break
                else:
                    break

        # 5. Wait for Reboot & Poll Until Back Online
        append_job_log(job_id, "Device is expanding packages and reloading. Waiting 90 seconds before polling...", "RELOADING")
        time.sleep(90)

        reconnected = False
        for attempt in range(1, 31):
            append_job_log(job_id, f"Polling SSH reachability (attempt {attempt}/30)...")
            try:
                with ConnectHandler(**conn_params) as check_conn:
                    check_conn.enable()
                    new_ver = check_conn.send_command("show version | include Cisco IOS")
                    append_job_log(job_id, f"Device is BACK ONLINE! New Version: {new_ver.strip()}")
                    reconnected = True
                    break
            except Exception:
                time.sleep(30)

        if not reconnected:
            append_job_log(job_id, "WARNING: Timed out waiting for device to return via SSH after 15 minutes.", "FAILED")
            return

        # 6. Optional Post-Upgrade Health Check
        if req.auto_post_check:
            append_job_log(job_id, "Waiting 20 seconds for interfaces/MACs to settle, then capturing POST snapshot...")
            time.sleep(20)
            post_res = collect_and_save_snapshot(
                req.host, req.username, req.password, req.secret, req.device_type, "POST"
            )
            append_job_log(job_id, f"POST snapshot saved at {post_res['timestamp_pst']}.")

        append_job_log(job_id, "IOS-XE Upgrade workflow finished successfully!", "COMPLETED")

    except Exception as e:
        append_job_log(job_id, f"Upgrade failed with error: {str(e)}", "FAILED")

@app.get("/api/images")
def list_images():
    items = []
    for fname in sorted(os.listdir(IMAGES_DIR)):
        fpath = os.path.join(IMAGES_DIR, fname)
        if os.path.isfile(fpath) and not fname.startswith("."):
            size_mb = round(os.path.getsize(fpath) / (1024 * 1024), 2)
            items.append({"filename": fname, "size_mb": size_mb})
    return items


@app.post("/api/images/upload")
async def upload_image(file: UploadFile = File(...)):
    safe_name = os.path.basename(file.filename)
    dest = os.path.join(IMAGES_DIR, safe_name)
    with open(dest, "wb") as out:
        while chunk := await file.read(1024 * 1024):
            out.write(chunk)
    return {"status": "uploaded", "filename": safe_name}


@app.post("/api/upgrade")
def start_upgrade(req: UpgradeRequest):
    conn = get_db()
    cur = conn.cursor()
    cur.execute(
        """
        INSERT INTO upgrade_jobs (device_ip, image_filename, status, logs)
        VALUES (%s, %s, 'RUNNING', %s)
        RETURNING id;
        """,
        (req.host, req.image_filename, f"Starting {req.mode} job for {req.host} -> {req.image_filename}\n"),
    )
    job_id = cur.fetchone()["id"]
    conn.commit()
    cur.close()
    conn.close()

    t = threading.Thread(target=execute_iosxe_upgrade, args=(job_id, req), daemon=True)
    t.start()
    return {"job_id": job_id, "status": "RUNNING"}

@app.get("/api/devices")
def list_devices():
    conn = get_db()
    cur = conn.cursor()
    cur.execute(
        """
        SELECT d.id, d.hostname, d.ip_address, d.device_type,
               MAX(CASE WHEN h.phase = 'PRE' THEN h.captured_at END) AS last_pre,
               MAX(CASE WHEN h.phase = 'POST' THEN h.captured_at END) AS last_post
        FROM devices d
        LEFT JOIN health_snapshots h ON d.id = h.device_id
        GROUP BY d.id
        ORDER BY d.id DESC;
        """
    )
    rows = cur.fetchall()
    cur.close()
    conn.close()

    for row in rows:
        if row["last_pre"]:
            row["last_pre"] = row["last_pre"].astimezone(PST).strftime("%Y-%m-%d %H:%M:%S %Z")
        if row["last_post"]:
            row["last_post"] = row["last_post"].astimezone(PST).strftime("%Y-%m-%d %H:%M:%S %Z")
    return rows


@app.post("/api/healthcheck")
def run_health_check(req: CheckRequest):
    if req.phase.upper() not in ("PRE", "POST"):
        raise HTTPException(status_code=400, detail="Phase must be 'PRE' or 'POST'")
    try:
        res = collect_and_save_snapshot(
            req.host, req.username, req.password, req.secret, req.device_type, req.phase
        )
        return {"status": "success", **res}
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"SSH Collection Failed: {str(e)}")


@app.get("/api/snapshots/{device_id}")
def get_device_snapshots(device_id: int):
    conn = get_db()
    cur = conn.cursor()
    cur.execute("SELECT * FROM devices WHERE id = %s;", (device_id,))
    device = cur.fetchone()
    if not device:
        cur.close()
        conn.close()
        raise HTTPException(status_code=404, detail="Device not found")

    cur.execute(
        "SELECT * FROM health_snapshots WHERE device_id = %s AND phase = 'PRE' ORDER BY captured_at DESC LIMIT 1;",
        (device_id,),
    )
    pre = cur.fetchone()

    cur.execute(
        "SELECT * FROM health_snapshots WHERE device_id = %s AND phase = 'POST' ORDER BY captured_at DESC LIMIT 1;",
        (device_id,),
    )
    post = cur.fetchone()
    cur.close()
    conn.close()

    def format_snap(snap):
        if not snap:
            return None
        return {
            "id": snap["id"],
            "timestamp_pst": snap["captured_at"].astimezone(PST).strftime("%Y-%m-%d %H:%M:%S %Z"),
            "interfaces": snap["interfaces_json"] or [],
            "arp_table": snap["arp_table_json"] or [],
            "mac_table": snap["mac_table_json"] or [],
            "running_config": snap["running_config"] or "",
        }

    config_diff = []
    if pre and post:
        config_diff = list(
            difflib.unified_diff(
                (pre["running_config"] or "").splitlines(),
                (post["running_config"] or "").splitlines(),
                fromfile="BEFORE_CONFIG",
                tofile="AFTER_CONFIG",
                lineterm="",
            )
        )

    return {
        "device": device,
        "before": format_snap(pre),
        "after": format_snap(post),
        "config_diff": config_diff,
    }
