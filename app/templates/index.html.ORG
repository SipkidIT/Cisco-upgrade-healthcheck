const CHART_COLORS = [
  '#38bdf8', '#4ade80', '#fbbf24', '#f87171', '#a78bfa',
  '#f472b6', '#2dd4bf', '#fb923c', '#818cf8', '#94a3b8'
];

function escapeHtml(val) {
  if (val === null || val === undefined) return '';
  return String(val)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;');
}

function normalizePort(name) {
  if (!name) return '';
  return String(name).trim().toLowerCase()
    .replace(/^twentyfivegige/, 'twe')
    .replace(/^hundredgige/, 'hu')
    .replace(/^fortygigabitethernet/, 'fo')
    .replace(/^tengigabitethernet/, 'te')
    .replace(/^gigabitethernet/, 'gi')
    .replace(/^fastethernet/, 'fa')
    .replace(/^ethernet/, 'eth')
    .replace(/^port-channel/, 'po');
}

function normalizeMac(mac) {
  return String(mac || '').toLowerCase().replace(/[^a-f0-9]/g, '');
}

async function loadImages() {
  const select = document.getElementById('upgImage');
  try {
    const res = await fetch('/api/images');
    if (!res.ok) throw new Error('Failed to load images');
    const items = await res.json();
    if (!items.length) {
      select.innerHTML = '<option value="">No .bin images uploaded yet</option>';
      return;
    }
    select.innerHTML = items.map(img =>
      `<option value="${escapeHtml(img.filename)}">${escapeHtml(img.filename)} (${img.size_mb} MB)</option>`
    ).join('');
  } catch (err) {
    select.innerHTML = '<option value="">Error loading images</option>';
  }
}

document.getElementById('uploadBinBtn').addEventListener('click', async () => {
  const fileInput = document.getElementById('binFileInput');
  const statusEl = document.getElementById('uploadStatus');
  const btn = document.getElementById('uploadBinBtn');

  if (!fileInput.files.length) {
    statusEl.style.color = '#f87171';

  }

  const formData = new FormData();

  btn.disabled = true;
  statusEl.style.color = '#38bdf8';
  statusEl.innerText = 'Uploading firmware to server repository...';

let pollTimer = null;

async function launchUpgrade(mode) {
  const host = document.getElementById('upgHost').value.trim();
  const username = document.getElementById('upgUser').value.trim();
  const password = document.getElementById('upgPass').value;
  const secret = document.getElementById('upgSecret').value;
  const device_type = document.getElementById('upgType').value;
  const image_filename = document.getElementById('upgImage').value;
  const expected_md5 = document.getElementById('upgMd5').value.trim();
  const auto_clean_flash = document.getElementById('autoCleanFlash').checked;
  const auto_pre_check = document.getElementById('autoPre').checked;
  const auto_post_check = document.getElementById('autoPost').checked;

  if (!host || !username || !password || !image_filename) {
    alert('Please enter Device IP, Username, Password, and select a Firmware Image.');
    return;
  }

  if (mode !== 'STAGE_ONLY') {
    const confirmed = confirm(`WARNING: '${mode}' will install ${image_filename} and REBOOT ${host}. Continue?`);
    if (!confirmed) return;
  }

  const consoleBox = document.getElementById('upgradeConsole');
  consoleBox.style.display = 'block';
  consoleBox.innerText = 'Submitting job to server...\n';

  try {
    const res = await fetch('/api/upgrade', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        host, username, password, secret, device_type,
        image_filename, expected_md5, mode, auto_clean_flash,
        auto_pre_check, auto_post_check
      })
    });
    const data = await res.json();
    if (!res.ok) throw new Error(data.detail || 'Failed to start job');

    if (pollTimer) clearInterval(pollTimer);
    pollTimer = setInterval(() => pollUpgradeStatus(data.job_id), 4000);
    pollUpgradeStatus(data.job_id);
  } catch (err) {
    consoleBox.innerText += `ERROR: ${err.message}\n`;
  }
}

async function pollUpgradeStatus(jobId) {
  const consoleBox = document.getElementById('upgradeConsole');
  try {
    const res = await fetch(`/api/upgrade/${jobId}`);
    if (!res.ok) return;
    const job = await res.json();
    consoleBox.innerText = job.logs || '';
    consoleBox.scrollTop = consoleBox.scrollHeight;

    if (job.status === 'COMPLETED' || job.status === 'FAILED') {
      clearInterval(pollTimer);
      pollTimer = null;
      await loadDevices();
    }
  } catch (err) {
    // Ignore transient network hiccup during polling
  }
}

function buildVendorPieChart(snap) {
  const sourceList = (snap.mac_table && snap.mac_table.length) ? snap.mac_table : (snap.arp_table || []);
  const sourceLabel = (snap.mac_table && snap.mac_table.length) ? 'MAC Table' : 'ARP Table';
  if (!sourceList.length) return '';

  const counts = {};
  sourceList.forEach(item => {
    const vendor = (item.vendor && item.vendor.trim()) ? item.vendor.trim() : 'Unknown';
    counts[vendor] = (counts[vendor] || 0) + 1;
  });

  const sorted = Object.entries(counts).sort((a, b) => b[1] - a[1]);
  const total = sourceList.length;

  let cumulativePct = 0;
  const gradientStops = [];
  const legendItems = [];

  sorted.forEach(([vendor, count], idx) => {
    const color = CHART_COLORS[idx % CHART_COLORS.length];
    const pct = (count / total) * 100;
    const start = cumulativePct.toFixed(2);
    cumulativePct += pct;
    const end = cumulativePct.toFixed(2);

    gradientStops.push(`${color} ${start}% ${end}%`);
    legendItems.push(`
      <li>
        <span class="legend-label" title="${escapeHtml(vendor)}">
          <span class="legend-dot" style="background:${color};"></span>
          ${escapeHtml(vendor)}
        </span>
        <span><strong>${pct.toFixed(1)}%</strong> (${count})</span>
      </li>
    `);
  });

  return `
    <h4>Vendor Distribution (${sourceLabel} &mdash; ${total} Total)</h4>
    <div class="chart-wrapper">
      <div class="pie-circle" style="background: conic-gradient(${gradientStops.join(', ')});"></div>
      <ul class="chart-legend">${legendItems.join('')}</ul>
    </div>
  `;
}

function indexSnapshotPorts(snap) {
  const map = {};
  if (!snap) return map;

  (snap.interfaces || []).forEach(iface => {
    const portLabel = iface.port || iface.interface || '';
    const key = normalizePort(portLabel);
    if (!key) return;
    map[key] = {
      port: portLabel,
      name: String(iface.name || '').trim(),
      status: String(iface.status || '').trim(),
      vlan: String(iface.vlan || '').trim(),
      duplex: String(iface.duplex || '').trim(),
      speed: String(iface.speed || '').trim(),
      type: String(iface.type || '').trim(),
      macs: []
    };
  });

  (snap.mac_table || []).forEach(m => {
    const rawPorts = Array.isArray(m.destination_port) ? m.destination_port : [m.destination_port || m.ports || ''];
    rawPorts.forEach(p => {
      const key = normalizePort(p);
      if (!key) return;
      if (!map[key]) {
        map[key] = {
          port: p,
          name: '',
          status: '',
          vlan: String(m.vlan || m.vlan_id || '').trim(),
          duplex: '',
          speed: '',
          type: 'Logical',
          macs: []
        };
      }
      if (!map[key].vlan && (m.vlan || m.vlan_id)) {
        map[key].vlan = String(m.vlan || m.vlan_id).trim();
      }
      map[key].macs.push({
        mac: String(m.destination_address || m.mac || '').trim(),
        normMac: normalizeMac(m.destination_address || m.mac || ''),
        vendor: String(m.vendor || 'Unknown').trim()
      });
    });
  });

  Object.values(map).forEach(entry => {
    entry.macs.sort((a, b) => a.normMac.localeCompare(b.normMac));
  });

  return map;
}

function buildCombinedInterfaceMacTable(snap, otherSnap, isPost) {
  const interfaces = snap.interfaces || [];
  const macs = snap.mac_table || [];
  if (!interfaces.length) {
    return '<p style="color:#94a3b8;">No interface data captured.</p>';
  }

  const currentMap = indexSnapshotPorts(snap);
  const otherMap = otherSnap ? indexSnapshotPorts(otherSnap) : null;
  const isSwitchStatus = interfaces[0].hasOwnProperty('port') || interfaces[0].hasOwnProperty('vlan');

  const legendHtml = otherMap ? `
    <div class="legend-bar">
      <span class="legend-chip"><span class="swatch" style="background:rgba(245,158,11,0.4);border:1px solid #f59e0b;"></span> Changed Value</span>
      <span class="legend-chip"><span class="swatch" style="background:rgba(34,197,94,0.35);"></span> Added</span>
      <span class="legend-chip"><span class="swatch" style="background:rgba(239,68,68,0.35);"></span> Missing After Upgrade</span>
    </div>
  ` : '';

  const rowsHtml = Object.entries(currentMap).map(([key, row]) => {
    const otherRow = otherMap ? otherMap[key] : null;
    const portMissingInOther = otherMap && !otherRow;

    let portClass = '';
    if (portMissingInOther) {
      portClass = isPost ? 'cell-added' : 'cell-missing';
    }

    const diffClass = (field) => {
      if (!otherRow) return '';
      return row[field] !== otherRow[field] ? 'cell-changed' : '';
    };

    const statusClass = /connect|up/i.test(row.status) && !/notconnect|down/i.test(row.status) ? 'status-up' : 'status-down';

    const otherMacMap = {};
    if (otherRow) {
      otherRow.macs.forEach(om => { otherMacMap[om.normMac] = om; });
    }

    const macEntriesHtml = [];
    const vendorEntriesHtml = [];

    row.macs.forEach(m => {
      let macCls = '';
      let vendorCls = '';

      if (otherRow) {
        const match = otherMacMap[m.normMac];
        if (!match) {
          macCls = isPost ? 'cell-added' : 'cell-missing';
          vendorCls = isPost ? 'cell-added' : 'cell-missing';
        } else if (match.vendor !== m.vendor) {
          vendorCls = 'cell-changed';
        }
      } else if (portMissingInOther) {
        macCls = isPost ? 'cell-added' : 'cell-missing';
        vendorCls = isPost ? 'cell-added' : 'cell-missing';
      }

      macEntriesHtml.push(`<div class="mac-entry ${macCls}"><code>${escapeHtml(m.mac)}</code></div>`);
      vendorEntriesHtml.push(`<div class="mac-entry ${vendorCls}"><strong>${escapeHtml(m.vendor)}</strong></div>`);
    });

    const macCell = macEntriesHtml.length ? macEntriesHtml.join('') : '<span style="color:#64748b;">&mdash;</span>';
    const vendorCell = vendorEntriesHtml.length ? vendorEntriesHtml.join('') : '<span style="color:#64748b;">&mdash;</span>';

    if (isSwitchStatus) {
      return `
        <tr>
          <td class="${portClass}"><strong>${escapeHtml(row.port)}</strong></td>
          <td class="${diffClass('name')}">${escapeHtml(row.name)}</td>
          <td class="${statusClass} ${diffClass('status')}">${escapeHtml(row.status)}</td>
          <td class="${diffClass('vlan')}">${escapeHtml(row.vlan)}</td>
          <td class="${diffClass('duplex')}">${escapeHtml(row.duplex)}</td>
          <td class="${diffClass('speed')}">${escapeHtml(row.speed)}</td>
          <td class="${diffClass('type')}">${escapeHtml(row.type)}</td>
          <td>${macCell}</td>
          <td>${vendorCell}</td>
        </tr>
      `;
    } else {
      return `
        <tr>
          <td class="${portClass}"><strong>${escapeHtml(row.port)}</strong></td>
          <td class="${statusClass} ${diffClass('status')}">${escapeHtml(row.status)}</td>
          <td>${macCell}</td>
          <td>${vendorCell}</td>
        </tr>
      `;
    }
  }).join('');

  const headersHtml = isSwitchStatus
    ? `<tr>
         <th>Port</th>
         <th>Name</th>
         <th>Status</th>
         <th>VLAN</th>
         <th>Duplex</th>
         <th>Speed</th>
         <th>Type</th>
         <th>MAC Address</th>
         <th>Vendor</th>
       </tr>`
    : `<tr>
         <th>Port</th>
         <th>Status</th>
         <th>MAC Address</th>
         <th>Vendor</th>
       </tr>`;

  return `
    <h4>Interface &amp; MAC Status (${interfaces.length} Ports / ${macs.length} MACs)</h4>
    ${legendHtml}
    <div style="overflow-x:auto;">
      <table>
        <thead>${headersHtml}</thead>
        <tbody>${rowsHtml}</tbody>
      </table>
    </div>
  `;
}

async function loadDevices() {
  const container = document.getElementById('deviceList');
  try {
    const res = await fetch('/api/devices');
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const devices = await res.json();
    if (!devices.length) {
      container.innerHTML = '<span style="color:#94a3b8;">No devices captured yet. Run a check above.</span>';
      return;
    }
    container.innerHTML = devices.map(d => `
      <div class="device-pill" onclick="loadComparison(${d.id})">
        <strong>${escapeHtml(d.hostname)}</strong> (${escapeHtml(d.ip_address)}) [${escapeHtml(d.device_type.toUpperCase())}]<br/>
        <small>Before: ${escapeHtml(d.last_pre || 'None')} | After: ${escapeHtml(d.last_post || 'None')}</small>
      </div>
    `).join('');
  } catch (err) {
    container.innerHTML = `<span style="color:#f87171;">Failed to load devices: ${escapeHtml(err.message)}</span>`;
  }
}

function renderSnapshot(snap, otherSnap, isPost) {
  if (!snap) return '<p style="color:#94a3b8;">Not captured yet.</p>';

  const pieChartHtml = buildVendorPieChart(snap);
  const combinedTableHtml = buildCombinedInterfaceMacTable(snap, otherSnap, isPost);

  const arpRows = snap.arp_table.map(a => `
    <tr>
      <td>${escapeHtml(a.address || a.ip_address || '')}</td>
      <td><code>${escapeHtml(a.mac || a.mac_address || '')}</code></td>
      <td><strong>${escapeHtml(a.vendor || 'Unknown')}</strong></td>
      <td>${escapeHtml(a.interface || '')}</td>
    </tr>
  `).join('');

  return `
    ${pieChartHtml}
    ${combinedTableHtml}

    <h4 style="margin-top:16px;">ARP Table with Vendor (${snap.arp_table.length})</h4>
    <div style="overflow-x:auto;">
      <table>
        <thead><tr><th>IP Address</th><th>MAC Address</th><th>Vendor</th><th>Interface</th></tr></thead>
        <tbody>${arpRows || '<tr><td colspan="4">No ARP entries</td></tr>'}</tbody>
      </table>
    </div>

    <h4 style="margin-top:16px;">Saved Running Configuration</h4>
    <pre>${escapeHtml(snap.running_config)}</pre>
  `;
}

async function loadComparison(deviceId) {
  const res = await fetch(`/api/snapshots/${deviceId}`);
  const data = await res.json();

  document.getElementById('comparisonArea').style.display = 'block';
  document.getElementById('compTitle').innerText = `${data.device.hostname} (${data.device.ip_address})`;
  document.getElementById('preTime').innerText = data.before ? data.before.timestamp_pst : '';
  document.getElementById('postTime').innerText = data.after ? data.after.timestamp_pst : '';

  document.getElementById('preContent').innerHTML = renderSnapshot(data.before, data.after, false);
  document.getElementById('postContent').innerHTML = renderSnapshot(data.after, data.before, true);

  const diffEl = document.getElementById('configDiff');
  if (data.before && data.after) {
    if (!data.config_diff.length) {
      diffEl.innerHTML = '<span style="color:#4ade80;">No configuration differences detected between BEFORE and AFTER.</span>';
    } else {
      diffEl.innerHTML = data.config_diff.map(line => {
        if (line.startsWith('+')) return `<div class="diff-add">${escapeHtml(line)}</div>`;
        if (line.startsWith('-')) return `<div class="diff-del">${escapeHtml(line)}</div>`;
        return `<div>${escapeHtml(line)}</div>`;
      }).join('');
    }
  } else {
    diffEl.innerText = 'Capture both BEFORE and AFTER snapshots to view line-by-line config changes.';
  }
}

    btn.disabled = false;
  }
});
